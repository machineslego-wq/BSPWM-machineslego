# Sway — Wayland-композитор и его окружение (переезд с BSPWM, требует sudo)
# Устанавливается скриптом scripts/install-sway.sh; конфиги не трогаются.
#
# Ставится поверх базовой установки: requirements/apt.txt (шрифты, иконки,
# Qt-темы, сеть, звук, jq, brightnessctl). Из базового набора сюда не
# дублируются, но конфиг Sway их вызывает — искать пакеты здесь:
#   network-manager       — nmcli и nmtui (wifi wofi, алиас net)
#   nm-connection-editor  — клик по модулю сети в waybar
#   brightnessctl         — модуль яркости в waybar
#   bluez                 — bluetoothctl (модуль Bluetooth в waybar, алиас bt)
#   jq                    — sway-display и sway-record
#   wireplumber           — wpctl (модуль звука в waybar)
#   qt6ct                 — QT_QPA_PLATFORMTHEME в sway-session
#   papirus-icon-theme    — иконки wofi
# power-profiles-daemon — из requirements/apt-system.txt (модуль профиля питания)

# Композитор и базовые компоненты
sway
swaybg
swayidle
swaylock

# Панель, лаунчер, терминал
# quickshell — панель, лаунчер и центр управления (config/quickshell/shell).
# В Debian 13 пакет есть только в trixie-backports: без них панель не
# поставится, и остаётся waybar. waybar и wofi — запасные (одна строка
# в config/sway/config; wofi ещё и фолбэк лаунчера в qs-shell).
quickshell
waybar
wofi
foot

# Qt-приложения под Wayland (sway-session выставляет QT_QPA_PLATFORM="wayland;xcb")
qt6-wayland
qtwayland5

# Уведомления. mako — запасной вариант: на ветке Quickshell уведомления
# рисует само приложение (NotificationServer), mako в config/sway выключен.
mako-notifier

# Скриншоты, запись экрана и буфер обмена
grim
slurp
wl-clipboard
swappy
wf-recorder
cliphist

# Портал скринкастов и X11-приложения
xdg-desktop-portal
xdg-desktop-portal-wlr
xdg-desktop-portal-gtk
xwayland

lxpolkit
